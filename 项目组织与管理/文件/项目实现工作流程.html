import trimesh import numpy as np def load_and_normalize(file_path): mesh = trimesh.load(file_path) # 1. 归一化：居中 + 缩放到单位球 mesh.vertices -= mesh.center_mass scale = np.max(np.linalg.norm(mesh.vertices, axis=1)) mesh.vertices /= scale # 2. 采样点云（2048个点） points, _ = trimesh.sample.sample_surface(mesh, 2048) return points # shape: (2048, 3)

方案一：传统手工特征（D2形状分布）—— 快速入门
适合资源有限
▾
原理：随机采样模型表面大量点对，计算点间距离分布直方图，构成"几何指纹"特征向量。
import numpy as np def d2_shape_distribution(points, n_bins=64, n_samples=100000): """D2形状分布特征提取""" idx1 = np.random.choice(len(points), n_samples) idx2 = np.random.choice(len(points), n_samples) # 计算采样点对之间的距离 dists = np.linalg.norm(points[idx1] - points[idx2], axis=1) # 归一化直方图作为特征 hist, _ = np.histogram(dists, bins=n_bins, range=(0, 2.0), density=True) return hist # 64维特征向量 # 批量提取所有模型的特征 all_points = np.load('data/processed/points.npy') features = np.array([d2_shape_distribution(pts) for pts in all_points]) np.save('features/d2_features.npy', features)

import torch from pointnet_cls import PointNetCls # 加载预训练模型（在ModelNet40上预训练） model = PointNetCls(k=40) model.load_state_dict(torch.load('pointnet_cls.pth')) def extract_pointnet_feature(points): """提取PointNet全局特征向量（1024维）""" model.eval() pts = torch.tensor(points).unsqueeze(0).transpose(2,1).float() with torch.no_grad(): _, feature, _ = model(pts) return feature.squeeze().numpy() # 1024维
✅ PointNet 提取的 1024 维特征向量，可直接用于分类（接全连接层）和相似性检索（计算余弦距离），一套特征两用，是整个系统的核心"指纹"。

import faiss import numpy as np features = np.load('features/pointnet_features.npy').astype('float32') # 归一化（余弦相似度） faiss.normalize_L2(features) # 建立 FAISS 索引 dim = features.shape[1] # 1024 index = faiss.IndexFlatIP(dim) # 内积 = 余弦相似度（归一化后） index.add(features) faiss.write_index(index, 'models/faiss.index') # 查询：找最相似的Top-10 def search_similar(query_feature, top_k=10): q = query_feature.reshape(1, -1).astype('float32') faiss.normalize_L2(q) scores, indices = index.search(q, top_k) return indices[0], scores[0]
